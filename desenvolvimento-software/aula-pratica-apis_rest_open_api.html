function renderizarSwagger() {
  const yamlContent = document.getElementById("editor-yaml").value; // textarea ou monaco
  const errorContainer = document.getElementById("error-container");
  const swaggerTarget = document.getElementById("swagger-ui-preview");

  // Limpa erros anteriores
  errorContainer.classList.add("hidden");
  errorContainer.textContent = "";

  let parsedSpec;

  // 1. Validação de Sintaxe YAML
  try {
    parsedSpec = jsyaml.load(yamlContent);
  } catch (err) {
    exibirErro(`Erro de sintaxe YAML (Linha ${err.mark?.line + 1 || '?'}, Coluna ${err.mark?.column + 1 || '?'}):\n${err.reason || err.message}`);
    return;
  }

  // 2. Validação Básica da Estrutura OpenAPI
  if (!parsedSpec || typeof parsedSpec !== "object") {
    exibirErro("O documento fornecido está vazio ou não contém uma estrutura YAML válida.");
    return;
  }

  if (!parsedSpec.openapi && !parsedSpec.swagger) {
    exibirErro("Campo obrigatório ausente: O contrato deve especificar a versão raiz (ex: 'openapi: 3.0.0' ou 'swagger: \"2.0\"').");
    return;
  }

  if (!parsedSpec.info || !parsedSpec.info.title || !parsedSpec.info.version) {
    exibirErro("Estrutura inválida em 'info': Os campos 'info.title' e 'info.version' são obrigatórios.");
    return;
  }

  if (!parsedSpec.paths || typeof parsedSpec.paths !== "object") {
    exibirErro("Campo 'paths' obrigatório ausente ou mal formatado.");
    return;
  }

  // 3. Renderização no Swagger UI
  try {
    swaggerTarget.innerHTML = ""; // Limpa a renderização antiga
    SwaggerUIBundle({
      spec: parsedSpec,
      dom_id: "#swagger-ui-preview",
      deepLinking: true,
      presets: [
        SwaggerUIBundle.presets.apis,
        SwaggerUIBundle.SwaggerUIStandalonePreset
      ],
      layout: "BaseLayout"
    });
  } catch (renderErr) {
    exibirErro(`Falha ao renderizar especificação no Swagger UI:\n${renderErr.message}`);
  }
}

function exibirErro(mensagem) {
  const errorContainer = document.getElementById("error-container");
  errorContainer.textContent = mensagem;
  errorContainer.classList.remove("hidden");
}

// Vincula a ação ao botão
document.getElementById("btn-renderizar").addEventListener("click", renderizarSwagger);
