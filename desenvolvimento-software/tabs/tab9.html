using System;

namespace LojaApi.Modules.Auth.Domain;

/// <summary>
/// Entidade Usuario do contexto Auth.
/// Didático: senha em texto puro. Não usar em produção real.
/// </summary>
public class Usuario
{
    /// <summary>Identificador do usuário. Vira o claim "sub" do JWT.</summary>
    public Guid Id { get; private set; }

    /// <summary>Login usado na autenticação. Deve ser único.</summary>
    public string Login { get; private set; } = string.Empty;

    /// <summary>Nome exibido no token e nas respostas.</summary>
    public string Nome { get; private set; } = string.Empty;

    /// <summary>
    /// Senha em texto puro (didático).
    /// Em produção real: hash forte, nunca texto puro.
    /// </summary>
    public string Senha { get; private set; } = string.Empty;

    /// <summary>Data de criação do usuário.</summary>
    public DateTime CriadoEm { get; private set; }

    /// <summary>
    /// Construtor protegido para o EF Core materializar a entidade.
    /// </summary>
    protected Usuario() { }

    /// <summary>
    /// Construtor principal com validações mínimas de domínio.
    /// </summary>
    public Usuario(Guid id, string login, string nome, string senha)
    {
        // Valida login: não pode ser vazio
        if (string.IsNullOrWhiteSpace(login))
            throw new ArgumentException("O login é obrigatório.");

        // Valida nome: não pode ser vazio
        if (string.IsNullOrWhiteSpace(nome))
            throw new ArgumentException("O nome é obrigatório.");

        // Valida senha: não pode ser vazia
        if (string.IsNullOrWhiteSpace(senha))
            throw new ArgumentException("A senha é obrigatória.");

        // Atribui os valores normalizando o login para minúsculas
        Id = id;
        Login = login.Trim().ToLower();
        Nome = nome.Trim();
        Senha = senha;
        CriadoEm = DateTime.UtcNow;
    }

    /// <summary>
    /// Compara a senha recebida com a senha armazenada.
    /// Didático: comparação direta em texto puro.
    /// </summary>
    public bool SenhaConfere(string senha)
    {
        // Se a senha enviada for nula/vazia, falha imediatamente
        if (string.IsNullOrWhiteSpace(senha))
            return false;

        // Comparação direta (texto puro)
        return Senha == senha;
    }
}

/// <summary>
/// Implementação MySQL/EF Core da porta IUsuarioRepository.
/// Para migrar para MongoDB no futuro, basta criar MongoUsuarioRepository
/// implementando a mesma interface e trocar o registro no DI.
/// </summary>
public class UsuarioRepository : IUsuarioRepository
{
    // DbContext do contexto Auth
    private readonly AuthDbContext _db;

    /// <summary>
    /// Injeção de dependência via construtor.
    /// </summary>
    public UsuarioRepository(AuthDbContext db)
    {
        _db = db;
    }

    /// <summary>
    /// Busca usuário pelo login.
    /// </summary>
    public async Task<Usuario?> BuscarPorLoginAsync(string login)
    {
        // Normaliza o login para minúsculas, igual ao construtor da entidade
        var loginNormalizado = login.Trim().ToLower();

        // Busca no banco usando AsNoTracking porque é somente leitura
        return await _db.Usuarios
            .AsNoTracking()
            .FirstOrDefaultAsync(u => u.Login == loginNormalizado);
    }

    /// <summary>
    /// Salva usuário no banco.
    /// </summary>
    public async Task<Usuario> SalvarAsync(Usuario usuario)
    {
        // Se o Id estiver vazio, trata como novo registro
        if (usuario.Id == Guid.Empty)
        {
            await _db.Usuarios.AddAsync(usuario);
        }
        else
        {
            // Caso contrário, atualiza o registro existente
            _db.Usuarios.Update(usuario);
        }

        // Persiste no banco
        await _db.SaveChangesAsync();

        // Retorna a entidade salva
        return usuario;
    }
}

LojaApi.Modules.Auth/Services/JwtService.cs 
using LojaApi.Modules.Auth.Domain;
using Microsoft.IdentityModel.Tokens;
using System;
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;

namespace LojaApi.Modules.Auth.Services;

/// <summary>
/// Gera JWT real usando HMAC-SHA256.
/// O token contém apenas:
/// - sub: id do usuário
/// - name: nome do usuário
/// - exp: expiração
/// </summary>
public class JwtService : IJwtService
{
    // Segredo usado para assinar o token
    private readonly string _secret;

    // Tempo de vida do token em minutos
    private readonly int _expirationMinutes;

    /// <summary>
    /// Construtor recebe secret e expiração.
    /// </summary>
    public JwtService(string secret, int expirationMinutes)
    {
        // Valida se o secret foi informado
        if (string.IsNullOrWhiteSpace(secret))
            throw new ArgumentException("O JWT Secret não pode ser vazio.");

        // HMAC-SHA256 exige chave com pelo menos 32 caracteres
        if (secret.Length < 32)
            throw new ArgumentException("O JWT Secret deve ter pelo menos 32 caracteres.");

        // Valida expiração positiva
        if (expirationMinutes <= 0)
            throw new ArgumentException("A expiração do JWT deve ser maior que zero.");

        _secret = secret;
        _expirationMinutes = expirationMinutes;
    }

    /// <summary>
    /// Gera o token JWT para o usuário.
    /// </summary>
    public string GerarToken(Usuario usuario)
    {
        // Converte o secret em chave simétrica
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_secret));

        // Cria as credenciais de assinatura
        var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        // Define as claims do token
        var claims = new[]
        {
            // sub: identificador do usuário
            new Claim(JwtRegisteredClaimNames.Sub, usuario.Id.ToString()),

            // name: nome do usuário
            new Claim("name", usuario.Nome),

            // login: útil para diagnóstico e testes
            new Claim("login", usuario.Login)
        };

        // Define a expiração do token
        var expiration = DateTime.UtcNow.AddMinutes(_expirationMinutes);

        // Monta o token JWT
        var token = new JwtSecurityToken(
            claims: claims,
            expires: expiration,
            signingCredentials: credentials
        );

        // Serializa o token para o formato string
        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}

using LojaApi.Modules.Auth.Interfaces;
using System.Threading.Tasks;

namespace LojaApi.Modules.Auth.Services;

/// <summary>
/// Caso de uso de login.
/// Depende da interface IUsuarioRepository, não da implementação MySQL.
/// Isso deixa o módulo pronto para futura troca por MongoDB.
/// </summary>
public class AuthService : IAuthService
{
    // Porta de acesso aos usuários
    private readonly IUsuarioRepository _usuarioRepository;

    // Serviço gerador de JWT
    private readonly IJwtService _jwtService;

    /// <summary>
    /// Injeção de dependência via construtor.
    /// </summary>
    public AuthService(IUsuarioRepository usuarioRepository, IJwtService jwtService)
    {
        _usuarioRepository = usuarioRepository;
        _jwtService = jwtService;
    }

    /// <summary>
    /// Valida credenciais e emite JWT.
    /// </summary>
    public async Task<LoginResult?> AutenticarAsync(string login, string senha)
    {
        // Se login ou senha estiverem vazios, falha imediatamente
        if (string.IsNullOrWhiteSpace(login) || string.IsNullOrWhiteSpace(senha))
            return null;

        // Busca o usuário pelo login
        var usuario = await _usuarioRepository.BuscarPorLoginAsync(login);

        // Se não encontrou o usuário, credenciais inválidas
        if (usuario is null)
            return null;

        // Compara senha em texto puro (didático)
        if (!usuario.SenhaConfere(senha))
            return null;

        // Gera o token JWT
        var token = _jwtService.GerarToken(usuario);

        // Retorna token + usuário autenticado
        return new LoginResult(
            token,
            new UsuarioAutenticado(usuario.Id, usuario.Nome, usuario.Login)
        );
    }
}

    public string Nome { get; private set; } = string.Empty;
    public decimal Preco { get; private set; }
    public int Estoque { get; private set; }
    public int Reservado { get; private set; }

    public int Disponivel => Estoque - Reservado;

    /// <summary>Construtor protegido para o EF Core.</summary>
    protected Produto() { }

    /// <summary>Construtor principal.</summary>
    public Produto(int id, string nome, decimal preco, int estoque)

        Id = id;
        Nome = nome;
        Preco = preco;
        Estoque = estoque;
    }

    /// <summary>
    /// Reserva quantidade do estoque.
    /// Lança DomainException 409 se estoque insuficiente.
    /// </summary>
    public void Reservar(int qtd)
    {
        // Valida quantidade positiva
        if (qtd <= 0)
            throw new DomainException("Quantidade deve ser > 0");

        // Valida estoque disponível
        if (Disponivel < qtd)
            throw new DomainException(
                $"Estoque insuficiente p/ produto {Id} (disp: {Disponivel}, ped: {qtd})",
                ProblemTypes.EstoqueInsuficiente);

        // Incrementa reservado
        Reservado += qtd;
    }

    /// <summary>Libera quantidade reservada (compensação).</summary>
    public void Liberar(int qtd)
    {
        Reservado = Math.Max(0, Reservado - qtd);
    }

    /// <summary>Confirma reserva: reduz estoque e libera reservado.</summary>
    public void ConfirmarReserva(int qtd)
    {
        Estoque -= qtd;
        Liberar(qtd);
    }

    /// <summary>
    /// Atualização validada (PUT do contrato).
    /// Invariantes de nome/preço/estoque.
    /// </summary>
    public void Atualizar(string nome, decimal preco, int estoque)
    {
        // Valida nome

            throw new DomainException("Nome deve ter >= 2 caracteres");

        // Valida preço positivo
        if (preco <= 0)
            throw new DomainException("Preço deve ser > 0");

        // Valida estoque não-negativo
        if (estoque < 0)
            throw new DomainException("Estoque não pode ser negativo");

        // Aplica atualização
        Nome = nome.Trim();
        Preco = preco;
        Estoque = estoque;
    }

    /// <summary>Construtor com configuração de timeout.</summary>

        _http.Timeout = TimeSpan.FromSeconds(2);
    }

    /// <summary>Reserva estoque no Catálogo.</summary>
    public async Task<ReservaDto> ReservarAsync(
        int produtoId,
        ReservarDto dto,
        CancellationToken ct = default)
    {
        // Faz POST para o endpoint de reserva
        var resp = await _http.PostAsJsonAsync(
            $"/api/catalogo/produtos/{produtoId}/reservar",
            dto,
            ct);

        // Se 409, estoque insuficiente
        if (resp.StatusCode == System.Net.HttpStatusCode.Conflict)
        {
            throw new DomainException(
                $"Estoque insuficiente p/ produto {produtoId}",
                ProblemTypes.EstoqueInsuficiente);

        if (!resp.IsSuccessStatusCode)
        {
            throw new DomainException(
                $"Catálogo indisponível ({(int)resp.StatusCode})",
                ProblemTypes.DependenciaIndisponivel);
        }

        // Desserializa resposta
        return (await resp.Content.ReadFromJsonAsync<ReservaDto>(ct))!;
    }

    public async Task LiberarAsync(
        int produtoId,
        ReservarDto dto,
        CancellationToken ct = default)

        try

                $"/api/catalogo/produtos/{produtoId}/liberar",

/// Caso de uso de checkout: reserva síncrona + validação
/// de endereço + compensação.
/// Sem alterações em relação à Tab 9 anterior.

public class PedidoService
{
    // Porta de saída para o Catálogo
    private readonly ICatalogoClient _catalogo;

    // Porta de saída para os Clientes
    private readonly IClientesClient _clientes;

    // DbContext do contexto Pedidos
    private readonly PedidosDbContext _db;

    /// <summary>Injeção de dependência via construtor.</summary>
    public PedidoService(
        ICatalogoClient catalogo,
        IClientesClient clientes,
        PedidosDbContext db)
    {
        _catalogo = catalogo;
        _clientes = clientes;
        _db = db;
    }

    /// <summary>
    /// Cria pedido: reserva estoque, valida endereço,
    /// persiste pedido. Compensa em caso de falha.
    /// </summary>
    public async Task<Pedido> CriarAsync(
        PedidoRequest req,
        string usuario,
        CancellationToken ct = default)
    {
        // Lista de reservas feitas (para compensação)
        var reservas = new List<(int ProdutoId, ReservarDto Dto)>();

        try
        {
            // Lista de itens do pedido
            var itens = new List<ItemPedido>();

            // Para cada item, reserva estoque no Catálogo
            foreach (var item in req.Itens)
            {
                // Cria DTO de reserva
                var dto = new ReservarDto(
                    item.Quantidade,
                    $"PED-{usuario}");

                // Reserva estoque
                var r = await _catalogo.ReservarAsync(
                    item.ProdutoId,
                    dto,
                    ct);

                // Registra reserva para compensação futura
                reservas.Add((item.ProdutoId, dto));

                // Cria item com snapshot de preço
                itens.Add(new ItemPedido(
                    item.ProdutoId,
                    item.Quantidade,
                    r.PrecoUnitario));
            }

            // Valida endereço do cliente
            var enderecos = await _clientes.ObterEnderecosAsync(
                req.ClienteId,
                ct);

            // Verifica se o endereço pertence ao cliente
            if (!enderecos.Any(e => e.Id == req.EnderecoId))
                throw new DomainException(
                    "Endereço não pertence ao cliente");

            // Cria pedido com factory method
            var pedido = Pedido.Criar(
                usuario,
                req.ClienteId,
                req.EnderecoId,
                itens);

            // Marca como reservado
            pedido.MarcarReservado(
                reservas.Select((_, i) => $"RES-{i:D4}"));

            // Persiste pedido
            _db.Pedidos.Add(pedido);
            await _db.SaveChangesAsync(ct);

            // Retorna pedido criado
            return pedido;
        }
        catch
        {
            // Compensação da saga-lite:
            // libera tudo que foi reservado
            foreach (var (produtoId, dto) in reservas)
                await _catalogo.LiberarAsync(produtoId, dto, ct);

            // Propaga o erro
            throw;
        }
    }

    /// <summary>
    /// Cancela pedido e libera reservas.
    /// </summary>
    public async Task CancelarAsync(
        int pedidoId,
        string usuario,
        CancellationToken ct = default)
    {
        // Busca pedido
        var p = await _db.Pedidos.FindAsync(
            new object[] { pedidoId },
            ct)
            ?? throw new DomainException("Pedido inexistente");

        // Valida que o pedido pertence ao usuário
        if (p.Usuario != usuario)
            throw new DomainException("Pedido de outro usuário");

        // Cancela pedido (valida transição de estado)
        p.Cancelar();

        // Libera reservas de estoque
        foreach (var i in p.Itens)
            await _catalogo.LiberarAsync(
                i.ProdutoId,
                new ReservarDto(i.Quantidade, $"PED-{pedidoId}"),
                ct);

        // Persiste no banco
        await _db.SaveChangesAsync(ct);
    }
}

// PASSO 1: Criar o builder da aplicação web
// ──────────────────────────────────────────────────────────────
var builder = WebApplication.CreateBuilder(args);

// ──────────────────────────────────────────────────────────────
// PASSO 2: Registrar serviços básicos
// ──────────────────────────────────────────────────────────────
// Controllers dos 4 módulos

// Swagger para documentação
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

// ──────────────────────────────────────────────────────────────
// PASSO 3: Ler configurações do appsettings.json
// ──────────────────────────────────────────────────────────────
// Connection strings (1 por contexto)
var authCs = builder.Configuration.GetConnectionString("Auth");
var catalogoCs = builder.Configuration.GetConnectionString("Catalogo");
var pedidosCs = builder.Configuration.GetConnectionString("Pedidos");
var clientesCs = builder.Configuration.GetConnectionString("Clientes");

// JWT
var jwtSecret = builder.Configuration["Jwt:Secret"];
var jwtExpiration = int.Parse(builder.Configuration["Jwt:ExpirationMinutes"] ?? "60");

// Endereços base para chamadas internas (Pedidos → Catálogo/Clientes)
var catalogoBase = new Uri(builder.Configuration["Services:Catalogo"] ?? "http://localhost:5000");
var clientesBase = new Uri(builder.Configuration["Services:Clientes"] ?? "http://localhost:5000");

// ──────────────────────────────────────────────────────────────
// PASSO 4: Registrar os 4 módulos (cada um registra suas deps)
// ──────────────────────────────────────────────────────────────
// 🔑 Auth: login, JWT, usuarios
builder.Services.AddAuthModule(authCs, jwtSecret, jwtExpiration);

// 📦 Catálogo: produtos, reserva de estoque
builder.Services.AddCatalogoModule(catalogoCs);

// 🛒 Pedidos: checkout, orquestração, clients HTTP
builder.Services.AddPedidosModule(pedidosCs, catalogoBase, clientesBase);

// 🧾 Clientes: cadastro, endereços
builder.Services.AddClientesModule(clientesCs);

// ──────────────────────────────────────────────────────────────
// PASSO 5: Configurar JWT Bearer SEM ROLES
// ──────────────────────────────────────────────────────────────
// Qualquer token válido é aceito. Sem verificação de roles.
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.TokenValidationParameters = new TokenValidationParameters
        {
            // Não valida issuer nem audience (didático)
            ValidateIssuer = false,
            ValidateAudience = false,

            // Valida expiração e assinatura
            ValidateLifetime = true,
            ValidateIssuerSigningKey = true,

            // Chave simétrica (mesma usada para gerar o token)
            IssuerSigningKey = new SymmetricSecurityKey(
                Encoding.UTF8.GetBytes(jwtSecret)),

            // Tolerância de 1 minuto para expiração
            ClockSkew = TimeSpan.FromMinutes(1),

            // Mapeia o claim "name" para User.Identity.Name
            // Assim, User.Identity.Name retorna o nome do usuário
            NameClaimType = "name"
        };
    });

// ──────────────────────────────────────────────────────────────

🚨 ExceptionMiddleware — RFC 7807

LojaApi.Api/Middleware/ExceptionMiddleware.cs 
using LojaApi.Shared.Kernel;
using System.Text.Json;

namespace LojaApi.Api.Middleware;

/// <summary>
/// Middleware centralizado de tratamento de exceções.
/// Converte DomainException em RFC 7807 (application/problem+json).
/// Sem roles: qualquer erro de domínio vira 400 ou 409.
/// </summary>
public class ExceptionMiddleware
{
    // Próximo middleware no pipeline
    private readonly RequestDelegate _next;

    /// <summary>Construtor padrão do middleware.</summary>
    public ExceptionMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    /// <summary>
    /// Intercepta toda request. Se DomainException for lançada,
    /// converte para RFC 7807.
    /// </summary>
    public async Task Invoke(HttpContext ctx)
    {
        try
        {
            // Passa para o próximo middleware
            await _next(ctx);
        }
        catch (DomainException ex)
        {
            // Determina o status code baseado no type da exceção
            var statusCode = GetStatusCode(ex.Type);

            // Configura resposta como RFC 7807
            ctx.Response.StatusCode = statusCode;
            ctx.Response.ContentType = "application/problem+json";

            // Serializa o ProblemDetails
            var problem = new
            {
                type = ex.Type,
                title = "Regra de negócio violada",
                status = statusCode,
                detail = ex.Message
            };

            // Escreve a resposta
            await ctx.Response.WriteAsync(
                JsonSerializer.Serialize(problem));
        }
    }

    /// <summary>
    /// Determina o status code baseado no type da exceção.
    /// </summary>
    private static int GetStatusCode(string type)
    {
        // Estoque insuficiente ou status inválido → 409 Conflict
        if (type.Contains("estoque-insuficiente") ||
            type.Contains("status-invalido"))
            return 409;

        // Dependência indisponível → 503 Service Unavailable
        if (type.Contains("dependencia-indisponivel"))
            return 503;

        // Qualquer outro erro de domínio → 400 Bad Request
        return 400;
    }
}
